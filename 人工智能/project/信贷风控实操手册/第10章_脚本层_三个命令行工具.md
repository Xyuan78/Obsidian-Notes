### 展开讲：为什么合成数据要"有逻辑"

如果标签是纯随机的（`y = random 0/1`），模型学不到任何东西，AUC 会稳定在 0.5。你会以为代码有 bug，其实是数据没信息。

所以合成数据的标签必须**由特征决定**：

```python
# 违约倾向 = 各风险因子的加权组合
risk = (
    0.5 * (dti 高)            # 负债收入比高 → 风险高
  + 0.4 * (FICO 低)           # 信用分低 → 风险高
  + 0.3 * (利率高)            # 利率高通常意味着评级差
  + 0.2 * (逾期次数多)
  + 0.2 * (循环额度用满)
  + 噪声
)
y = 1 if risk > 阈值 else 0
```

这样造出来的数据，XGBoost 能学到真实规律，AUC 会落在 0.72~0.78——和真实数据接近。

### 步骤 10.1　写 `scripts/generate_sample_data.py`

**【目标】** 生成字段完整、有内在规律的合成数据。

**【文件】** `scripts/generate_sample_data.py`（新建）

**【完整代码】**

```python
#!/usr/bin/env python
"""
合成数据生成器
==============
生成与天池"贷款违约预测"字段一致的假数据。
标签由风险因子按逻辑构造，所以模型能真正学到东西。
"""
from __future__ import annotations

import argparse
import sys
from pathlib import Path

import numpy as np
import pandas as pd

PROJECT_ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(PROJECT_ROOT))


def generate_sample_data(n_samples: int = 50000, random_seed: int = 42) -> pd.DataFrame:
    rng = np.random.default_rng(random_seed)
    df = pd.DataFrame()
    df["id"] = np.arange(n_samples)

    # ---- 贷款基本信息 ----
    df["loanAmnt"] = np.round(rng.lognormal(9.5, 0.6, n_samples)).clip(500, 40000)
    df["term"] = rng.choice([3, 5], size=n_samples, p=[0.7, 0.3])
    df["interestRate"] = np.round(rng.normal(13, 4, n_samples).clip(5, 30), 2)

    # 月还款：等额本息公式
    monthly_rate = df["interestRate"] / 100 / 12
    n_months = df["term"] * 12
    df["installment"] = np.round(
        df["loanAmnt"] * monthly_rate * (1 + monthly_rate) ** n_months
        / ((1 + monthly_rate) ** n_months - 1),
        2,
    )

    # ---- 客户画像 ----
    df["grade"] = rng.choice(
        ["A", "B", "C", "D", "E", "F", "G"],
        size=n_samples, p=[0.15, 0.25, 0.25, 0.18, 0.10, 0.05, 0.02],
    )
    df["subGrade"] = df["grade"] + rng.integers(1, 6, n_samples).astype(str)
    df["employmentTitle"] = rng.integers(0, 250000, n_samples).astype(float)
    df["employmentLength"] = rng.choice(
        ["< 1 year", "1 year", "2 years", "3 years", "4 years", "5 years",
         "6 years", "7 years", "8 years", "9 years", "10+ years"],
        size=n_samples,
        p=[0.10, 0.07, 0.10, 0.09, 0.07, 0.07, 0.05, 0.05, 0.05, 0.05, 0.30],
    )
    df["homeOwnership"] = rng.choice([0, 1, 2, 3], size=n_samples, p=[0.40, 0.50, 0.08, 0.02])
    df["annualIncome"] = np.round(rng.lognormal(11, 0.6, n_samples), 2).clip(8000, 1500000)
    df["verificationStatus"] = rng.choice([0, 1, 2], size=n_samples, p=[0.30, 0.40, 0.30])

    # ---- 时间 ----
    df["issueDate"] = pd.to_datetime(
        rng.choice(pd.date_range("2014-01-01", "2019-12-31"), size=n_samples)
    ).strftime("%Y-%m-%d")
    df["earliesCreditLine"] = (
        pd.to_datetime(df["issueDate"])
        - pd.to_timedelta(rng.integers(365 * 5, 365 * 30, n_samples), unit="D")
    ).dt.strftime("%b-%Y")

    # ---- 其他 ----
    df["purpose"] = rng.integers(0, 14, n_samples)
    df["postCode"] = rng.integers(0, 1000, n_samples).astype(float)
    df["regionCode"] = rng.integers(0, 50, n_samples)
    df["dti"] = np.round(rng.normal(18, 8, n_samples).clip(0, 60), 2)
    df["delinquency_2years"] = rng.poisson(0.3, n_samples).astype(float)
    df["ficoRangeLow"] = rng.normal(690, 30, n_samples).clip(580, 850).round().astype(int)
    df["ficoRangeHigh"] = df["ficoRangeLow"] + 4
    df["openAcc"] = rng.poisson(11, n_samples).astype(float)
    df["pubRec"] = rng.poisson(0.2, n_samples).astype(float)
    df["pubRecBankruptcies"] = rng.poisson(0.1, n_samples).astype(float)
    df["revolBal"] = np.round(rng.lognormal(8.5, 1.2, n_samples), 2)
    df["revolUtil"] = np.round(rng.beta(2, 3, n_samples) * 100, 2)
    df["totalAcc"] = rng.poisson(25, n_samples).astype(float) + 1
    df["initialListStatus"] = rng.choice([0, 1], n_samples, p=[0.6, 0.4])
    df["applicationType"] = rng.choice([0, 1], n_samples, p=[0.95, 0.05])
    df["title"] = rng.integers(0, 100, n_samples).astype(float)
    df["policyCode"] = 1.0

    # ---- 匿名特征 ----
    for i in range(15):
        df[f"n{i}"] = np.round(rng.normal(0, 1, n_samples), 3)

    # ===== 关键：按风险逻辑构造标签 =====
    # 各因子归一到 0~1，越接近 1 越危险
    dti_risk = np.clip(df["dti"] / 60, 0, 1)
    fico_risk = np.clip((750 - df["ficoRangeLow"]) / 170, 0, 1)
    rate_risk = np.clip((df["interestRate"] - 5) / 25, 0, 1)
    delinq_risk = np.clip(df["delinquency_2years"] / 5, 0, 1)
    util_risk = np.clip(df["revolUtil"] / 100, 0, 1)
    income_risk = np.clip((120000 - df["annualIncome"]) / 120000, 0, 1)
    burden_risk = np.clip(
        df["installment"] / (df["annualIncome"] / 12 + 1) / 0.5, 0, 1
    )

    linear = (
        1.6 * dti_risk
        + 1.9 * fico_risk
        + 1.2 * rate_risk
        + 1.0 * delinq_risk
        + 0.8 * util_risk
        + 0.9 * income_risk
        + 1.3 * burden_risk
        - 4.2                                   # 截距，控制整体违约率在 20% 左右
    )
    prob = 1 / (1 + np.exp(-linear))
    df["isDefault"] = rng.binomial(1, prob).astype(int)

    return df


def main():
    parser = argparse.ArgumentParser(description="生成合成训练数据")
    parser.add_argument("--n_samples", type=int, default=50000)
    parser.add_argument("--output", default="data/sample/sample.csv")
    parser.add_argument("--seed", type=int, default=42)
    args = parser.parse_args()

    df = generate_sample_data(args.n_samples, args.seed)

    out = Path(args.output)
    out.parent.mkdir(parents=True, exist_ok=True)
    df.to_csv(out, index=False, encoding="utf-8")

    print(f"已生成 {len(df):,} 行 × {df.shape[1]} 列 -> {out}")
    print(f"违约率: {df['isDefault'].mean():.2%}")
    print(f"前 5 列: {list(df.columns[:5])}")


if __name__ == "__main__":
    main()
```

### 步骤 10.2　写 `scripts/train_pipeline.py`

**【目标】** 命令行一键训练。

**【文件】** `scripts/train_pipeline.py`（新建）

**【完整代码】**

```python
#!/usr/bin/env python
"""
训练流水线
==========
用法:
    python scripts/train_pipeline.py --use-sample          # 合成数据，最快
    python scripts/train_pipeline.py                       # 真实数据
    python scripts/train_pipeline.py --use-sample --skip-mlp
    python scripts/train_pipeline.py --tune                # 启用超参优化（慢）
"""
from __future__ import annotations

import argparse
import sys
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(PROJECT_ROOT))

from src.pipeline.orchestrator import TrainingPipeline
from src.utils.config_loader import ConfigLoader
from src.utils.logger import get_logger


def main():
    parser = argparse.ArgumentParser(description="智能信贷风控分析系统 - 训练流水线")
    parser.add_argument("--config-dir", default=str(PROJECT_ROOT / "configs"))
    parser.add_argument("--use-sample", action="store_true", help="使用合成数据（快）")
    parser.add_argument("--skip-mlp", action="store_true", help="跳过 MLP 训练")
    parser.add_argument("--tune", action="store_true", help="启用超参优化（耗时）")
    args = parser.parse_args()

    logger = get_logger("scripts.train")
    logger.info(f"加载配置: {args.config_dir}")
    configs = ConfigLoader(args.config_dir)

    pipeline = TrainingPipeline(configs)
    result = pipeline.run(
        use_sample=args.use_sample,
        skip_mlp=args.skip_mlp,
        skip_tuning=not args.tune,
    )

    print("\n" + "=" * 60)
    print("训练完成 - 评估结果")
    print("=" * 60)
    print(f"运行 ID:   {result['run_id']}")
    print(f"耗时:      {result['duration']:.2f}s")
    print(f"训练样本:  {result['train_samples']:,}")
    print(f"特征数:    {result['feature_count']}")
    print()

    for name, m in result["eval_results"].items():
        print(f"[{name.upper()}]")
        print(f"  AUC:       {m['auc']:.4f}")
        print(f"  KS:        {m['ks']:.4f}")
        print(f"  F1:        {m['f1']:.4f}")
        print(f"  Accuracy:  {m['accuracy']:.4f}")
        print(f"  Precision: {m['precision']:.4f}")
        print(f"  Recall:    {m['recall']:.4f}")
        print()

    print("模型已保存至 data/models/latest/")
    print("接下来运行: streamlit run app/streamlit_app.py")


if __name__ == "__main__":
    main()
```

### 错误 W-07　合成数据和真实数据的字段格式不一致

**现象**　用合成数据训得好好的，换成真实 `train.csv` 后到处报错或指标崩到 0.5。

**原因**　两边同名字段的**类型和取值**不一样：

| 字段 | 合成数据 | 真实数据 |
|---|---|---|
| `term` | 整数 `3` / `5` | 字符串 `"36 months"` / `"60 months"` |
| `employmentLength` | `"< 1 year"` | `"10+ years"`、`"< 1 year"`（带 NaN） |
| `homeOwnership` | 整数 `0/1/2/3` | 字符串 `"RENT"`/`"OWN"`/`"MORTGAGE"` |
| `verificationStatus` | 整数 `0/1/2` | 字符串 `"Verified"` 等 |
| `grade` | `"A"`~`"G"` | 一致 |

**💥 不修的后果**　`_apply_encoders` 在整数列上做 `astype(str)`，编码结果与训练时完全不同，模型看到的特征全错位。不报错，但 AUC 会异常低。

**✅ 处理**　要么让合成数据完全模仿真实格式（`term` 改成 `"36 months"`），要么在 `DataLoader` 里加一层 schema 归一化：

```python
def _normalize_schema(self, df):
    if "term" in df.columns and df["term"].dtype == object:
        df["term"] = df["term"].str.extract(r"(\d+)").astype(float) / 12
    return df
```

**答辩提示**　"合成数据与真实数据的 schema 对齐"是个真实的工程问题，主动讲出来是加分项。

## 8. 本章自测

**1. 为什么脚本开头要 `sys.path.insert(0, PROJECT_ROOT)`？**

<details>
<summary>点开看答案</summary>

直接运行 `python scripts/xx.py` 时，Python 只把 `scripts/` 加进搜索路径，找不到同级的 `src/`。把项目根目录插进去，`import src.xxx` 才能成功。

</details>

**2. 合成数据的标签为什么不能随机生成？**

<details>
<summary>点开看答案</summary>

随机标签里没有可学的规律，模型 AUC 会稳定在 0.5，你会误以为代码有 bug。标签必须由特征按逻辑决定，模型才能学到东西。

</details>

**3. `--tune` 和不加 `--tune` 的区别？**

<details>
<summary>点开看答案</summary>

不加时 `skip_tuning=True`，直接用 YAML 里的默认超参训练，几十秒完成。加了才跑 Optuna 搜索，可能几十分钟。

</details>
