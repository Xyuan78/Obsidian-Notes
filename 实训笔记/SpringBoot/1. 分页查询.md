
## 1. 先建一个分页插件

在config（配置类）下面创建一个类MybatiesPlusConfig

```java
import com.baomidou.mybatisplus.annotation.DbType;
import com.baomidou.mybatisplus.extension.plugins.MybatisPlusInterceptor;
import com.baomidou.mybatisplus.extension.plugins.inner.PaginationInnerInterceptor;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

@Configuration // 必须加上这个注解，告诉 Spring 这是一个配置类
public class MybatisPlusConfig {

    @Bean
    public MybatisPlusInterceptor mybatisPlusInterceptor() {
        MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
        
        // 1. 实例化分页插件，并指定数据库类型为 MySQL
        PaginationInnerInterceptor pagination = new PaginationInnerInterceptor(DbType.MYSQL);
        
        // 2. 核心优化：单页分页最大条数限制(防止恶意请求拖垮数据库)
        pagination.setMaxLimit(100L);
        
        // 3. 还可以设置溢出总页数后是否进行处理 (默认 false)
        // pagination.setOverflow(false);
        
        // 4. 将分页插件添加到拦截器中
        interceptor.addInnerInterceptor(pagination);
        
        return interceptor;
    }
}
```
@Bean：创建一个对象（用在方法上），由spring管理对象

## 4. 在impl里面实现service的方法
```java
  @Override
    public IPage<Food> pageByCondition(String foodName,
                                       BigDecimal minPrice,
                                       BigDecimal maxPrice,
                                       int no,
                                       int size) {
        if (minPrice != null && maxPrice != null && minPrice.compareTo(maxPrice) > 0) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "最低价不能高于最高价");
        }

        int pageNo = Math.max(no, 1);
        int pageSize = Math.min(Math.max(size, 1), 50);

        String keyword = foodName == null ? "" : foodName.trim();

        QueryWrapper<Food> wrapper = new QueryWrapper<>();
        wrapper.like(StringUtils.hasText(keyword), "food_name", keyword)
                .ge(minPrice != null, "food_price", minPrice)
                .le(maxPrice != null, "food_price", maxPrice)
                .orderByAsc("food_id");

        Page<Food> page = new Page<>(pageNo, pageSize);
        IPage<Food> result = page(page, wrapper);

        log.info("分页查询：foodName={}, minPrice={}, maxPrice={}, no={}, size={}, total={}, pages={}",
                keyword, minPrice, maxPrice, pageNo, pageSize,
                result.getTotal(), result.getPages());

        return result;
    }
```
