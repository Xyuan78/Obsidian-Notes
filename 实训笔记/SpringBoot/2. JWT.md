## 2. 在util包下
编写配置,JwtTokenUtil,这里要定义一个加密的秘钥，用于jjwt的加密与解密，

```java
package com.qxy.demo1.util;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.ExpiredJwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.SignatureAlgorithm;

import java.util.Date;

/**
 * JWT 工具类：生成、解析、校验。
 * 课堂项目用静态方法，和生产项目的注入写法不同，但原理一样。
 */
public final class JwtTokenUtil {

    /** 请求头名字 */
    public static final String TOKEN_HEADER = "Authorization";

    /** 请求头前缀 */
    public static final String TOKEN_PREFIX = "Bearer ";

    /** 验证码 token 有效期：5 分钟 */
    public static final long CAPTCHA_EXPIRATION = 5 * 60 * 1000L;

    /** 登录 token 有效期：1 天 */
    public static final long LOGIN_EXPIRATION = 24 * 60 * 60 * 1000L;

    /**
     * 签名密钥，HS512 要求足够长（建议 64 字节以上）。
     * 正式项目应该放到配置文件/环境变量里，不要提交到仓库。
     */
    private static final String APP_SECRET =
            "qxy-food-jwt-secret-2026-change-me-to-a-long-random-string-0123456789abcdef";

    private static final String USER_NAME = "username";

    private JwtTokenUtil() {
    }

    /**
     * 生成 token。
     */
    public static String createToken(String username, long expiration) {
        Date now = new Date();
        return Jwts.builder()
                .setSubject(username)
                .claim(USER_NAME, username)
                .setIssuedAt(now)
                .setExpiration(new Date(now.getTime() + expiration))
                .signWith(SignatureAlgorithm.HS512, APP_SECRET)
                .compact();
    }

    private static Claims getTokenClaims(String token) {
        try {
            return Jwts.parser()
                    .setSigningKey(APP_SECRET)
                    .parseClaimsJws(token)
                    .getBody();
        } catch (ExpiredJwtException e) {
            return e.getClaims();
        } catch (Exception e) {
            return null;
        }
    }

    /**
     * 从 token 里取 username。
     */
    public static String getUserName(String token) {
        if (token == null || token.isEmpty()) {
            return null;
        }
        Claims claims = getTokenClaims(token);
        if (claims == null) {
            return null;
        }
        Object username = claims.get(USER_NAME);
        return username == null ? null : username.toString();
    }

    /**
     * 是否过期。
     */
    public static boolean isExpiration(String token) {
        Claims claims = getTokenClaims(token);
        return claims == null
                || claims.getExpiration() == null
                || claims.getExpiration().before(new Date());
    }

    /**
     * token 是否有效（能解析且没过期）。
     */
    public static boolean validate(String token) {
        if (token == null || token.isEmpty()) {
            return false;
        }
        Claims claims = getTokenClaims(token);
        return claims != null
                && claims.getExpiration() != null
                && claims.getExpiration().after(new Date());
    }
}
```
